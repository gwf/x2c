```x2c
#include <assert.h>
#include <string.h>
#include <stdlib.h>
#define FIELDS(X) X(alpha) X(beta)                 // X-macros expand
#define DECLARE(n) int n;
typedef struct Config { FIELDS(DECLARE) } Config;
~int main(void) {
Config c = { 1, 2 }, *view = &c;
view.beta = 5;                                     // . reaches through a pointer
assert(view->beta == 5 && c.alpha == 1);           // -> still works
char *dup = strdup("plain");                       // C strings, C allocation
String s = String.new(dup);                        // and back into x2c
assert(s == "plain" && strlen(s) == 5);
free(dup);
Var boxed = c.beta;                                // C scalars box into Var
assert(boxed is int);
~return 0;
~}
```

- Headers are headers, structs are the same bytes, libraries link. Pass
  `-I`, `-l`, and `-O2` to `x2c build` as you would to `cc`.
- `#ifdef`, token pasting, and macros that supply grammar are expanded
  before parsing. `x2c translate --cpp-symbols` shows what C sees.
- Some pointer mismatches C only warns about are errors here.
